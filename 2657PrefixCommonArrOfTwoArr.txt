OPTIMISED SOLUTION :

class Solution {
    public int[] findThePrefixCommonArray(int[] A, int[] B) {
        HashSet<Integer> setA = new HashSet<>();
        HashSet<Integer> setB = new HashSet<>();
        int count =0;
        int ans[] = new int[A.length];
        for(int i = 0; i<A.length; i++){
            if(A[i]==B[i])
                count++;
            else{
                if(setB.contains(A[i])){
                    count++;
                }
                 if(setA.contains(B[i])){
                    count++;
                }
            }
            setA.add(A[i]);
            setB.add(B[i]);
            ans[i] = count;

        }
        return ans;
    }
}

class Solution {
    public int[] findThePrefixCommonArray(int[] A, int[] B) {
        int n = A.length;
        int[] freq = new int[n + 1];
        int[] ans = new int[n];
        int count = 0;

        for(int i = 0; i < n; i++) {
            if(++freq[A[i]] == 2) {
                count++;
            }
            if(++freq[B[i]] == 2) {
                count++;
            }
            ans[i] = count;
        }
        return ans;
    }
}

C# =>
using System.Collections.Generic;

public class Solution
{
    public int[] FindThePrefixCommonArray(int[] A, int[] B)
    {
        HashSet<int> setA = new HashSet<int>();
        HashSet<int> setB = new HashSet<int>();
        int n = A.Length;
        int[] ans = new int[n];
        int count = 0;
        for(int i = 0; i < n; i++)
        {
            // Same element in both arrays
            if(A[i] == B[i])
            {
                count++;
            }
            else
            {
                // A[i] already appeared in B prefix
                if(setB.Contains(A[i]))
                {
                    count++;
                }
                // B[i] already appeared in A prefix
                if(setA.Contains(B[i]))
                {
                    count++;
                }
            }
            // Add current elements
            setA.Add(A[i]);
            setB.Add(B[i]);
            ans[i] = count;
        }
        return ans;
    }
}
