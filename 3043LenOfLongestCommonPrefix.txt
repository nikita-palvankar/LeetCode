OPTIMISED SOLUTION ->

class Solution {
    public int longestCommonPrefix(int[] arr1, int[] arr2) {
        //create hashset
        HashSet<Integer> set = new HashSet<>();
        //traverse arr1
        for(int num: arr1){
            //generate prefixes
            while(num > 0){
                set.add(num);
                num /= 10;
            }
        }
        //traverse arr2
        int max = 0;
        for(int num: arr2){
            //generate prefixes
            while(num > 0){
                if(set.contains(num)){
                    // if prefix exists find length
                    int length = String.valueOf(num).length();
                    // then find max length 
                    max = Math.max(max, length);
                    break;
                }
                num /= 10;
            }
        }

        return max;
    }
}

using System;
using System.Collections.Generic;

public class Solution {
    
    public int LongestCommonPrefix(int[] arr1, int[] arr2) {
        
        HashSet<int> set = new HashSet<int>();

        // Store all prefixes from arr1
        foreach(int num in arr1)
        {
            int temp = num;

            while(temp > 0)
            {
                set.Add(temp);
                temp = temp / 10;
            }
        }

        int max = 0;

        // Check prefixes from arr2
        foreach(int num in arr2)
        {
            int temp = num;

            while(temp > 0)
            {
                if(set.Contains(temp))
                {
                    int length = temp.ToString().Length;

                    max = Math.Max(max, length);

                    break; // first match is longest
                }

                temp = temp / 10;
            }
        }

        return max;
    }
}
