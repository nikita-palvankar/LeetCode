 # Character Occurrence from a String =>

public static int charOccurrence(String s, String c){
    int count = 0;

    for(int i = 0; i < s.length(); i++){
        String a = "" + s.charAt(i);

        if(c.equals(a)){
            count++;
        }
    }

    return count;
}

# Improved version ->

public static int charOccurrence(String s, String c){
    int count = 0;

    for(int i = 0; i < s.length(); i++){
        if(s.charAt(i) == c.charAt(0)){
            count++;
        }
    }

    return count;
}
